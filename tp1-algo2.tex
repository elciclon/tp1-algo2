		\\
		\obs{promo}{\Tipo{Promo}}
		\\
		\obs{promoConTope}{\Tipo{PromoConTope}}
		\\
		
		
		\begin{proc}{crearSistema}
			{
			}{
				\Tipo{AirMiles}
			}
			\requiereCorto{true}
			\asegura{res.usuarios = \Tipo{\{\}} \land res.Promo[nombre] = \text{´NoPromo´} \land res.Promo[factor] = 1}
			
		\end{proc}
		
		\begin{proc}{registrarUsuario}
			{\Inout aerolínea: \Tipo{AirMiles}, \In usuario:\Tipo{Usuario}}{}
			\requiere{usuario \notin aerolínea.usuarios \land aerolínea = aerolínea0}
			\asegura{\begin{aligned}
					& |aerolínea.usuarios| = |aerolínea0.usuarios| + 1 \ \land\\
					& aerolínea.usuarios = setKey(aerolínea0.usuarios, usuario,  \langle Vuelos:\vacio, MillasGanadas:\vacio,\\ 
					& MillasCanjeadas:\vacio, MillasRecibidas:\vacio, MillasTransferidas:\vacio\rangle)\ \land \\
					& aerolínea.promo = aerolínea0.promo \ \land \\
					& aerolínea.promoConTope = aerolínea0.promoConTope
					
			\end{aligned}}
		\end{proc}
		
		
		
		
		\newpage
		\begin{proc}{registrarUsuarioConReferidor}
			{\begin{tabular}[t]{@{}l@{}}
					{\Inout aerolínea: \Tipo{AirMiles},\\ \In usuario:\Tipo{Usuario},\\
						\In referidor:\Tipo{Usuario}}{}
			\end{tabular}}{}
			\requiere{aerolínea = aerolínea0 \land usuario \notin aerolínea.usuarios \land referidor \in aerolínea.usuarios}
			\asegura{\begin{aligned}                               
					& |aerolínea.usuarios| = |aerolínea0.usuarios| + 1 \ \land\\
					& aerolínea.usuarios = setKey(aerolínea0.usuarios, usuario,  \langle Vuelos:\vacio,\\
					& MillasGanadas: \langle Millas: 500,\\  
					& nombrePromo: aerolínea0.usuarios[referidor].MillasGanadas.nombrePromo, \\ 
					& factorPromo: aerolinea0.usuarios[referidor].MillasGanadas.factorPromo \rangle \\ 
					& MillasCanjeadas:\vacio, MillasRecibidas:\vacio, MillasTransferidas:\vacio\rangle\ \land \\
					& aerolínea.usuarios = setKey(agregarUsuario(aerolínea0.usuarios, usuario),\\ 
					& referidor, concat(aerolínea0.usuarios[referidor].MillasGanadas, \langle 500 \rangle))\ \land \\ 
					& aerolínea.promo = aerolínea0.promo \ \land \\
					& aerolínea.promoConTope = aerolínea0.promoConTope 
			\end{aligned}}
		\end{proc}
		
		\aux{agregarUsuario}{usuarios: \Tipo{Usuarios}, usuario: \Tipo{Usuario}}{Usuarios}
		{
			{\begin{aligned}   
					& setKey(usuarios, usuario,  \langle Vuelos:\vacio, MillasGanadas:\vacio MillasCanjeadas:\vacio,\\          
					& MillasRecibidas:\vacio, MillasTransferidas:\vacio\rangle)
		\end{aligned}}}
		
		
		\begin{proc}{acumularMillas}
			{\begin{tabular}[t]{@{}l@{}}
					{\Inout aerolínea: \Tipo{AirMiles},\\ \In usuario:\Tipo{Usuario},\\ \In distanciaVolada:\Tipo{Vuelo},\\
						\In promo:\Tipo{Promo}}{}
			\end{tabular}}{}
			
			\requiere{aerolínea = aerolínea0 \land usuario \in aerolínea.usuarios \land promo \in aerolínea.promo \land distanciaVolada > 0}
			\asegura{\begin{aligned}                               
					& aerolínea.usuarios = setKey(aerolínea0.usuarios, usuario, \\  
					& \langle concat(aerolínea0.usuarios[usuario].Vuelos, distanciaVolada),\\
					& concat(aerolínea0.usuarios[usuario].MillasGanadas,\\
					& agregarMillasGanadas(aerolínea0.usuarios[usuario].MillasGanadas)),\\
					& aerolíneas0.usuarios[usuario].MillasCanjeadas,\\  
					& aerolíneas0.usuarios[usuario].MillasRecibidas,\\   & aerolíneas0.usuarios[usuario].MillasTransferidas\rangle
					\ \land \\ 
					& aerolínea.promo = aerolínea0.promo \ \land \\
					& aerolínea.promoConTope = aerolínea0.promoConTope 
			\end{aligned}}
			
		\end{proc}
