					& aerolínea.promosConTope = aerolínea0.promosConTope
					
			\end{aligned}}
		\end{proc}

		\begin{proc}{acumularMillas}{
				\begin{tabular}[t]{@{}l@{}}
					\Inout aerolínea : \Tipo{AirMiles}, \\
					\In usuario : \Tipo{Usuario}, \\
					\In distanciaVolada : \Tipo{Vuelos}, \\
					\In promo : $\str$
				\end{tabular}
			}{}    
			\requiere{\begin{tabular}[t]{@{}l@{}}
					$aerolínea = aerolínea_0 \land usuario \in aerolínea.usuarios \land (promo \in aerolínea0.promos\ $\lor \\
					$promo \in aerolínea0.promosConTope) \ $\land$ \\
					$distanciaVolada > 0$
			\end{tabular}}
			\asegura{\begin{aligned}                               
					& aerolínea.usuarios = agregarNuevaTransacción(aerolínea0.usuarios,usuario,\\
					& distanciaVolada,promo,
					\text{IfThenElse}(promo\in aerolínea0.promos,\\
					& aerolínea0.promos[promo],
					aerolínea0.promosConTope[promo].factor) 
					\ \land \\ 
					& aerolínea.promosConTope = actualizarPromoConTope(promo,\\
					& aerolínea.usuarios[usuario], aerolíneas.promosConTope)\ \land\\
					& aerolíneas.promos = aerolíneas0.promos
			\end{aligned}}
			
		\end{proc}
		
		\newpage
		
		\aux{agregarNuevaTransacción}{usuarios: \Tipo{Usuarios}, usuario: \Tipo{Usuario},distanciaVolada : \Tipo{Vuelos},promo : \str, factor: \R}{Usuarios}{
			{\begin{aligned} 
					& setKey(usuarios, usuario,\langle vuelos: concat(aerolínea0.usuarios[usuario].vuelos, \\
					& \langle distanciaVolada\rangle),\\
					& millasGanadas:concat(usuarios[usuario].millasGanadas,\\
					& \langle agregarMillasGanadas(usuarios[usuario], distanciaVolada, promo)\rangle,\\
					& millasCanjeadas:usuarios[usuario].millasCanjeadas,\\  
					& millasRecibidas:usuarios[usuario].millasRecibidas,\\   
					& millasTransferidas:usuarios[usuario].millasTransferidas\rangle)\\
			\end{aligned}}
		}
		
		\aux{agregarMillasGanadas}{distanciaVolada : \Tipo{Vuelos}, transacciones : \Tipo{Transacciones}, \\promo : \str, factor:\R}{MillasGanadas}{
			{\begin{aligned}   
					& \langle millas:calcularMillasAcumuladas(distanciaVolada,transacciones,factor),\\
					& nombrePromo: promo, factorPromo: factor\rangle
			\end{aligned}}    
		}
		
		\aux{calcularMillasAcumuladas}{distanciaVolada : \Tipo{Vuelos}, transacciones : \Tipo{Transacciones},\\ factor : \R}{\R}
		{\begin{aligned}   
				& distanciaVolada * factorCategoria(transacciones) * factor
		\end{aligned}}
		
		\aux{factorCategoría}{transacciones : \Tipo{Transacciones}}{\R}{
			\begin{aligned}
				&\text{IfThenElse}(esBronce(millasTotales(transacciones)), 1, \\
				& \quad\text{IfThenElseFi}(esPlata(millasTotales(transacciones)), 1.5, \\
				&\quad\quad\text{IfThenElseFi}(esOro(millasTotales(transacciones)), 2, 0)))
			\end{aligned}
		}
		
		\pred{esBronce}{millasTotales: \R}
		{millasTotales<10000}
		
		\pred{esPlata}{millasTotales: \R}
		{10000\leq millasTotales<50000}
		
		\pred{esOro}{millasTotales: \R}
		{50000\leq millasTotales<1000000}
		
		
		
		\begin{proc}{transferirMillas}
			{\begin{tabular}[t]{@{}l@{}}
					{\Inout aerolínea : airMiles, \In usuarioTransferidor : \Tipo{Usuario},\\ \In usuarioReceptor : \Tipo{Usuario}, 
						\In millasATransferir : \Tipo{Millas}}{}
			\end{tabular}}{}
			\requiere{\begin{tabular}[t]{@{}l@{}}
					$aerolínea = aerolínea_0 \land usuarioTransferidor \in aerolínea.usuarios \ \land $ \\ 
					$usuarioReceptor \in aerolínea.usuarios \ \land $ \\
					$millasATransferir \leq millasDisponibles(aerolínea0.usuarios[usuarioTransferidor])
			\end{tabular}}
			\asegura{\begin{aligned}                               
					& aerolínea.usuarios = setKey(agregarTransferenciaRecibida(aerolínea0.usuarios, \\
					& usuarioTransferidor, usuarioReceptor, millasATransferir),  \\
					& usuarioTransferidor, \langle vuelos: aerolínea0.usuarios[usuarioReceptor].vuelos, \\
					& millasGanadas: aerolínea0.usuarios[usuarioTransferidor].millasGanadas, \\
					& millasCanjeadas: aerolínea0.usuarios[usuarioTransferidor].millasCanjeadas, \\
					& millasRecibidas:aerolíneas0.usuarios[usuarioTransferidor].millasRecibidas,\\  
					& millasTransferidas: \\ 
					& concat(aerolíneas0.usuarios[usuarioTransferidor].millasTransferidas, \\
					& \langle receptor: usuarioReceptor, millas: millasATransferir \rangle) \rangle)
					\ \land \\ 
					& aerolínea.promos = aerolínea0.promos \ \land \\
					& aerolínea.promosConTope = aerolínea0.promosConTope 
			\end{aligned}}
		\end{proc}
		
		
		\aux{agregarTransferenciaRecibida}
		{usuarios: \Tipo{Usuarios}, usuarioTransferidor: \Tipo{Usuario}, \\usuarioReceptor: \Tipo{Usuario}, millasATransferir: Millas}{\Tipo{Usuarios}}
		
		{
			{\begin{aligned}   
					& setKey(usuarios, usuarioReceptor, \langle vuelos: usuarios[usuarioReceptor].vuelos, \\ 
					& millasGanadas: usuarios[usuarioReceptor].millasGanadas, \\ 
					& millasCanjeadas: usuarios[usuarioReceptor].millasCanjeadas, \\  
					& millasRecibidas: concat(usuarios[usuarioReceptor].millasRecibidas, \\
					& \langle transferidor: usuarioTransferidor, millas: millasATransferir \rangle) \rangle) \\
					& millasTransferidas: usuarios[usuarioReceptor].millasTransferidas,\\
			\end{aligned}}\\\\}
		
		
		\begin{proc}{consultarSaldo}
			{\Inout aerolínea: \Tipo{AirMiles}, \In usuario: \Tipo{Usuario}}{\Tipo{Millas}}{}
			\requiere{usuario \in aerolínea.usuarios}
			\asegura{ res = millasDisponibles(aerolínea.usuarios[usuario])}
		\end{proc}
		
		\begin{proc}{rankingMillas}
			{\In aerolinea: \AirMiles, \in cantidad: \Z}{\seq<\usuario>}
			\requiere{cantidad \geq 0}
			\asegura{
				((|res|= cantidad) \lor (|res|=|aerolinea.usuario|\land cantidad=0) \land
				((\Forall usuario: \Z \land{\text{L}} usuario \in aerolinea.usuarios) \implies 
				(\exists n:\Z)(\Forall m: \Z)(0 \leq n,m < |res| \implies{\text{L}} (ifThenElse((n>m) \land( millasTotales(aerolinea.usuarios[n])>millasTotales(aerolinea.usuarios[m])),True,False)))))
			}
		\end{proc}
