		\begin{proc}{registrarUsuario}
			{\Inout aerolínea: \Tipo{AirMiles}, \In usuario:\Tipo{Usuario}}{}
			\requiere{usuario \notin aerolínea.usuarios \land aerolínea = aerolínea0}
			\asegura{\begin{aligned}
					% & |aerolínea.usuarios| = |aerolínea0.usuarios| + 1 \ \land\\
					& aerolínea.usuarios = setKey(aerolínea0.usuarios, usuario,  \langle vuelos:\vacio, millasGanadas:\vacio,\\ 
					& millasCanjeadas:\vacio, millasRecibidas:\vacio, millasTransferidas:\vacio\rangle)\ \land \\
					& aerolínea.promos = aerolínea0.promos \ \land \\
					& aerolínea.promosConTope = aerolínea0.promosConTope
					
			\end{aligned}}
		\end{proc}

		\begin{proc}{registrarUsuarioConReferidor}
			{\begin{tabular}[t]{@{}l@{}}
					{\Inout aerolínea: \Tipo{AirMiles},\\ \In usuario:\Tipo{Usuario},\\
						\In referidor:\Tipo{Usuario}}{}
			\end{tabular}}{}
			\requiere{aerolínea = aerolínea0 \land usuario \notin aerolínea.usuarios \land referidor \in aerolínea.usuarios}

					% & |aerolínea.usuarios| = |aerolínea0.usuarios| + 1 \ \land\\
					& aerolínea.usuarios = setKey(agregarUsuarioReferido(aerolínea0.usuarios, usuario),\\ 
					& referidor, \langle vuelos:aerolínea0.usuarios[referidor].vuelos,\\ 
					& millasGanadas: concat(aerolínea0.usuarios[referidor].millasGanadas,\\
					& \langle millas:500, nombrePromo:\text{'NoPromo'},factorPromo:1 \rangle),\\
					& millasCanjeadas:aerolínea0.usuarios[referidor].millasCanjeadas, \\
					& millasRecibidas:aerolíneas0.usuarios[referidor].millasRecibidas, \\

					& aerolínea.promosConTope = aerolínea0.promosConTope 
			\end{aligned}}
		\end{proc}
		
		\aux{agregarUsuarioReferido}{usuarios: \Tipo{Usuarios}, usuario: \Tipo{Usuario}}{Usuarios}
		{
			{\begin{aligned}   
					& setKey(usuarios, usuario,  \langle vuelos:\vacio,\\ 
					&millasGanadas:\langle millas:500, nombrePromo:\text{'NoPromo'},factorPromo:1 \rangle,\\ &millasCanjeadas:\vacio,  
					millasRecibidas:\vacio, millasTransferidas:\vacio\rangle)
			\end{aligned}}\\\\}
		
		
		\begin{proc}{acumularMillas}
			{\begin{tabular}[t]{@{}l@{}}
					{\Inout aerolínea: \Tipo{AirMiles},\\ \In usuario:\Tipo{Usuario},\\ \In distanciaVolada:\Tipo{Vuelos},\\
						\In promo:\Tipo{Promo}}{}
			\end{tabular}}{}
			
			\requiere{\begin{tabular}[t]{@{}l@{}}
					$aerolínea = aerolínea_0 \land usuario \in aerolínea.usuarios \land promo \in aerolínea.promos \ \land$ \\
					$distanciaVolada > 0$
			\end{tabular}}
			\asegura{\begin{aligned}                               
					& aerolínea.usuarios = setKey(aerolínea0.usuarios, usuario, \\        
					& \langle vuelos: concat(aerolínea0.usuarios[usuario].vuelos, \langle distanciaVolada\rangle),\\
					& millasGanadas:concat(aerolínea0.usuarios[usuario].millasGanadas,\\
					& \langle agregarMillasGanadas(aerolínea0.usuarios[usuario], distanciaVolada, promo)\rangle,\\
					& millasCanjeadas:aerolíneas0.usuarios[usuario].millasCanjeadas,\\  
					& millasRecibidas:aerolíneas0.usuarios[usuario].millasRecibidas,\\   
					& millasTransferidas:aerolíneas0.usuarios[usuario].millasTransferidas\rangle)
					\ \land \\ 
					& aerolínea.promos = aerolínea0.promos \ \land \\
					& aerolínea.promosConTope = aerolínea0.promosConTope 
			\end{aligned}}
			
		\end{proc}
		
		\aux{agregarMillasGanadas}{distanciaVolada:\Tipo{Vuelos}, transacciones:\Tipo{Transacciones}, promo:\Tipo{Promo}}{MillasGanadas}{
			{\begin{aligned}   
					& \langle millas:calcularMillasAcumuladas(distanciaVolada,transacciones,promo),\\
					& nombrePromo: promo.nombre, factorPromo: promo.factor\rangle
			\end{aligned}}    
			
		}
		
		\aux{calcularMillasAcumuladas}{distanciaVolada:\Tipo{Vuelos}, transacciones:\Tipo{Transacciones}, promo:\Tipo{Promo}}{\R}
		{\begin{aligned}   
				& distanciaVolada * factorCategoria(transacciones) * promo.factor
		\end{aligned}}
		
		\aux{factorCategoría}{transacciones: \Tipo{Transacciones}}{\R}{
			\begin{aligned}
				&\text{IfThenElse}(esBronce(millasTotales(transacciones)), 1, \\
				& \quad\text{IfThenElseFi}(esPlata(millasTotales(transacciones)), 1.5, \\
				&\quad\quad\text{IfThenElseFi}(esOro(millasTotales(transacciones)), 2, 0)))
			\end{aligned}
		}
		
		\pred{esBronce}{millasTotales: \R}
		{millasTotales<10000}
		
		\pred{esPlata}{millasTotales: \R}
		{10000\leq millasTotales<50000}
		
		\pred{esOro}{millasTotales: \R}
		{50000\leq millasTotales<1000000}
		
		
		\begin{proc}{canjearMillas}
			{\Inout airMiles, \In id: \Z, \In millasACanjear: \R}{}
			\requiere{(id \in usuarios[n]) \land millasACanjear > 0 \land usuarios0=usuarios \land promo0=promo}
			\aseguraLargo{
				airMiles.usuarios[id]++[("millasCanjeadas",millasACanjear)] \land 
				(\forall n: \Z)((0 \leq n < |airMiles.usuarios0[id]|)\implies{\text{L}} airMiles.usuarios[id][n] = airMiles.usuarios0[id][n])\land
				promo=promo0 \land
				(\forall i: \Z)(((i \in airMiles.usuarios) \land (i \neq id)) \implies{\text{L}} airMiles.usuarios[i] = airMiles.usuarios0[i])
			}
		\end{tad}
		
		\begin{proc}{rankingMillas}
			{\In aerolinea: \AirMiles, \in cantidad: \Z}{\seq<\usuario>}
			\requiere{cantidad \geq 0}
			\asegura{
				((|res|= cantidad) \lor (|res|=|aerolinea.usuario|\land cantidad=0) \land
				((\Forall usuario: \Z \land{\text{L}} usuario \in aerolinea.usuarios) \implies 
				(\exists n:\Z)(\Forall m: \Z)(0 \leq n,m < |res| \implies{\text{L}} (ifThenElse((n>m) \land( millasTotales(aerolinea.usuarios[n])>millasTotales(aerolinea.usuarios[m])),True,False)))))
			}
		\end{proc}
